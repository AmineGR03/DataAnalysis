% ============================================================
% SECTION 1 : INTRODUCTION ET GÉNÉRALITÉS
% ============================================================
% Cette section présente l'historique et les concepts fondamentaux
% d'Unix/Linux. Essentiel pour comprendre l'évolution et l'architecture
% des systèmes Unix modernes.

\section{Introduction et G\'en\'eralit\'es}

% ------------------------------------------------------------
% Sous-section : Histoire d'Unix
% ------------------------------------------------------------
% Comprendre l'historique permet de saisir l'évolution des systèmes
% Unix et leur importance dans l'informatique moderne. Les dates clés
% sont importantes pour l'examen.

\subsection{Histoire d'Unix}
\begin{itemize}
    \item \textbf{1969} : Cr\'eation d'UNIX par Ken Thompson aux laboratoires Bell
    \item \textbf{1973} : R\'e\'ecriture compl\`ete en langage C par Denis Ritchie
    \item \textbf{1974} : AT\&T propose les premi\`eres licences aux universit\'es
    \item \textbf{Ann\'ees 80} : Clonage d'UNIX par d'autres constructeurs (ULTRIX, BSD, AIX)
    \item \textbf{1983} : Lancement du projet GNU par Richard Stallman
    \item \textbf{1991} : Cr\'eation du noyau Linux par Linus Torvalds
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Linux - Définition
% ------------------------------------------------------------
% IMPORTANT : Distinction entre le noyau Linux et le système d'exploitation
% complet. Le noyau seul ne suffit pas, il faut les outils GNU pour
% former un système complet (d'où GNU/Linux).

\subsection{Linux : D\'efinition}
% Le noyau Linux est le cœur du système, gérant les ressources matérielles
% et les processus. Le système complet inclut également les outils utilisateur.
Linux est le noyau d'un OS de type Unix, initi\'e par Linus Torvalds en 1991. Il d\'esigne un OS comparable \`a Windows ou Mac OS compos\'e de nombreux logiciels libres.

% ------------------------------------------------------------
% Sous-section : Caractéristiques de Linux
% ------------------------------------------------------------
% Ces caractéristiques expliquent pourquoi Linux est utilisé dans
% de nombreux environnements (serveurs, supercalculateurs, embarqué).

\subsection{Caract\'eristiques de Linux}
% GPL (General Public License) : licence libre permettant modification
% et redistribution. POSIX : standard garantissant compatibilité entre
% systèmes Unix. SMP : Symmetric Multi-Processing, plusieurs processeurs
% partageant la même mémoire.
\begin{itemize}
    \item Code source disponible (licence GPL) % Permet audit et modification
    \item Syst\`eme multit\^ache et multi-utilisateur % Plusieurs tâches et utilisateurs simultanés
    \item Multi-plateforme (Intel x86, Sun Sparc, etc.) % Portabilité importante
    \item Gestion du multiprocesseur (SMP) % Performance sur serveurs
    \item Compatible POSIX % Standard Unix garantissant compatibilité
    \item Gestion des consoles virtuelles % Plusieurs sessions texte simultanées (Ctrl+Alt+F1-F6)
    \item Impl\'ementation compl\`ete de la pile TCP/IP % Réseau natif
    \item Interface graphique X-Window % Système de fenêtrage
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Architecture d'un système Linux
% ------------------------------------------------------------
% Architecture en couches : chaque couche communique avec les couches
% adjacentes. Le kernel fait le lien entre matériel et logiciel.

\subsection{Architecture d'un syst\`eme Linux}
% Modèle en couches : séparation des responsabilités
% 1. Matériel : disques, RAM, CPU, périphériques
% 2. Kernel : gestion des ressources, processus, mémoire, fichiers
% 3. Interface : outils utilisateur, shell, applications
Trois couches :
\begin{enumerate}
    \item \textbf{Couche physique} : p\'eriph\'eriques + BIOS % Matériel et firmware
    \item \textbf{Couche syst\`eme} : kernel et processus % Noyau gérant ressources
    \item \textbf{Couche interface} : shell + X-Window % Interaction utilisateur
\end{enumerate}

% ------------------------------------------------------------
% Sous-section : Le Shell
% ------------------------------------------------------------
% Le shell est l'interface en ligne de commande. Bash est le plus
% répandu aujourd'hui. Chaque shell a sa syntaxe et ses fonctionnalités.

\subsection{Le Shell}
% Shell = interpréteur de commandes. Transforme commandes texte en
% appels système. Bash (Bourne Again Shell) est le standard Linux.
% Différents shells : syntaxe et fonctionnalités varient.
Le Shell est l'interpr\'eteur de commande, interface entre l'utilisateur et l'OS. Types de shell disponibles : bash (sh), csh, ksh, zsh, etc.

% ============================================================
% SECTION 2 : STANDARD DE HIÉRARCHIE DES SYSTÈMES DE FICHIERS (FHS)
% ============================================================
% Le FHS standardise l'organisation des répertoires sous Linux/Unix.
% Connaître cette structure est essentiel pour l'administration système.
% Chaque répertoire a un rôle précis et des conventions à respecter.

\section{Standard de Hi\'erarchie des Syst\`emes de Fichiers (FHS)}

% ------------------------------------------------------------
% Sous-section : Définition du FHS
% ------------------------------------------------------------
% Standard maintenu par Linux Foundation. Garantit cohérence entre
% distributions et facilite administration et portabilité.

\subsection{D\'efinition}
% Le FHS définit où placer quels fichiers. Important pour :
% - Trouver rapidement fichiers de configuration
% - Comprendre structure système
% - Respecter conventions lors installation logiciels
Le FHS (Filesystem Hierarchy Standard) d\'efinit l'arborescence et le contenu des principaux r\'epertoires des syst\`emes de fichiers Linux/Unix.

\subsection{R\'epertoires principaux}

% ------------------------------------------------------------
% Sous-sous-section : /bin, /sbin, /lib
% ------------------------------------------------------------
% Distinction importante : /bin et /sbin sont dans la partition racine
% (nécessaires au démarrage minimal). /usr/bin et /usr/sbin peuvent
% être sur partition séparée.

\subsubsection{/bin, /sbin, /lib}
% /bin : commandes essentielles disponibles même en mode single-user
% /sbin : commandes système nécessitant privilèges root
% /lib : bibliothèques partagées (.so) utilisées par programmes
\begin{itemize}
    \item \textbf{/bin} : ex\'ecutables pour tous les utilisateurs (ls, cp, mv, vi, bash) % Commandes de base
    \item \textbf{/sbin} : ex\'ecutables pour administration (shutdown, ifconfig, fsck) % Nécessitent root
    \item \textbf{/lib} : biblioth\`eques partag\'ees (shared libraries) % .so files, partagées entre programmes
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : /dev
% ------------------------------------------------------------
% Les fichiers dans /dev sont des interfaces vers périphériques.
% Lecture/écriture sur ces fichiers = communication avec matériel.
% /dev/null : "poubelle", ignore tout. /dev/zero : source de zéros.

\subsubsection{/dev}
% Fichiers spéciaux : interface entre logiciel et matériel
% Block devices : accès par blocs (disques) - /dev/sda, /dev/sdb
% Character devices : accès caractère par caractère (terminaux) - /dev/tty
Contient des fichiers sp\'eciaux (device files) correspondant aux p\'eriph\'eriques.
\begin{itemize}
    \item Block device files (disques) % Accès par blocs, ex: /dev/sda1
    \item Character device files (terminaux) % Accès séquentiel, ex: /dev/tty1
    \item Commande : \texttt{mknod} pour cr\'eer des fichiers sp\'eciaux % Rarement utilisé directement
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : /etc
% ------------------------------------------------------------
% Répertoire de configuration système. Modifications nécessitent souvent
% root. Sauvegarder avant modifications importantes.

\subsubsection{/etc}
% Configuration système : fichiers texte modifiables par admin
% /etc/passwd, /etc/shadow, /etc/group : gestion utilisateurs
% /etc/fstab : montage automatique partitions
% /etc/hosts : résolution noms locaux
Contient les fichiers et scripts de configuration des diff\'erents services :
\begin{itemize}
    \item /etc/X11 : configuration X-window % Système graphique
    \item /etc/rc.d : scripts de d\'emarrage % Scripts init (SysV)
    \item /etc/cron : t\^aches p\'eriodiques % Cron jobs (crontab)
    \item /etc/skel : fichiers \`a recopier pour nouveaux utilisateurs % Template home directory
    \item /etc/sysconfig : configuration des p\'eriph\'eriques % Variables système
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : /usr
% ------------------------------------------------------------
% /usr peut être monté en lecture seule ou sur partition séparée.
% Contient programmes non essentiels au démarrage minimal.

\subsubsection{/usr}
% Unix System Resources : programmes utilisateur (non essentiels au boot)
% Peut être sur partition séparée (montage après /)
% /usr/local : installations locales (priorité sur /usr)
Unix System Resources - programmes et utilitaires non indispensables :
\begin{itemize}
    \item /usr/bin, /usr/sbin, /usr/lib : \'equivalents de /bin, /sbin, /lib % Programmes additionnels
    \item /usr/etc : fichiers de configuration (rarement utilis\'e) % Peu utilisé
    \item /usr/include : fichiers .h pour le compilateur C % Headers compilation
    \item /usr/local : arborescence propre \`a la machine % Installations locales, priorité
    \item /usr/share : fichiers ind\'ependants de l'architecture % Docs, man, données
    \item /usr/src : sources % Code source (optionnel)
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : /var
% ------------------------------------------------------------
% Données variables : taille change pendant fonctionnement système.
% Surveiller espace disque, surtout /var/log.

\subsubsection{/var}
% Variable data : fichiers qui changent pendant fonctionnement
% Surveiller espace : /var/log peut grandir rapidement
% /var/tmp : fichiers temporaires persistants (contrairement à /tmp)
Fichiers dont la taille peut cro\^itre :
\begin{itemize}
    \item /var/lock : fichiers de verrouillage % Évite accès concurrents
    \item /var/log : journaux du syst\`eme % Logs système et applications
    \item /var/spool : files d'attentes (cron, lpd, mail) % Files d'attente
    \item /var/run : fichiers contenant les PID des services % PID files, sockets
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : /proc
% ------------------------------------------------------------
% Interface vers kernel. Fichiers virtuels = pas sur disque.
% Permet consulter/modifier paramètres kernel sans redémarrer.

\subsubsection{/proc}
% Pseudo-filesystem : interface kernel en temps réel
% /proc/cpuinfo : infos CPU
% /proc/meminfo : infos mémoire
% /proc/PID/ : informations sur processus spécifique
% Modifications possibles : /proc/sys/ (sysctl)
Pseudo-syst\`eme de fichiers utilis\'e comme interface avec les structures de donn\'ees du noyau. Fichiers virtuels cr\'e\'es dynamiquement en m\'emoire.

\subsubsection{Autres r\'epertoires}
\begin{itemize}
    \item \textbf{/boot} : fichiers utiles pour le chargeur et noyaux Linux
    \item \textbf{/home} : espaces priv\'es des utilisateurs
    \item \textbf{/mnt} : points de montage des partitions externes
    \item \textbf{/media} : points de montage des unit\'es amovibles
    \item \textbf{/tmp} : fichiers temporaires
    \item \textbf{/root} : espace de travail de l'administrateur
    \item \textbf{/lost+found} : fichiers perdus et r\'ecup\'er\'es par fsck
\end{itemize}

% ============================================================
% SECTION 3 : DÉMARRAGE ET ARRÊT DU SYSTÈME LINUX
% ============================================================
% Comprendre le processus de démarrage est crucial pour le dépannage.
% Chaque étape peut être source de problèmes. Connaître les outils
% de diagnostic (journalctl, systemctl) est essentiel.

\section{D\'emarrage et Arr\^et du Syst\`eme Linux}

% ------------------------------------------------------------
% Sous-section : Processus de démarrage
% ------------------------------------------------------------
% Séquence critique : chaque étape doit réussir pour passer à la suivante.
% Problèmes fréquents : GRUB corrompu, fichiers système manquants,
% services défaillants.

\subsection{Processus de d\'emarrage}
% Séquence de boot : hardware -> firmware -> bootloader -> kernel -> init
% Chaque étape peut échouer : diagnostic par étape
\begin{enumerate}
    \item Power on
    \item BIOS/UEFI
    \item BootLoader (GRUB)
    \item Linux Kernel
    \item Processus PID 1 (init/systemd)
    \item System Ready
\end{enumerate}

% ------------------------------------------------------------
% Sous-section : BIOS vs UEFI
% ------------------------------------------------------------
% UEFI est le successeur du BIOS. Plus moderne, plus de fonctionnalités.
% GPT nécessaire pour disques > 2TB, supporté nativement par UEFI.

\subsection{BIOS vs UEFI}
% BIOS : ancien standard, limitations (MBR, 2TB max)
% UEFI : moderne, Secure Boot, GPT natif, interface graphique
\begin{itemize}
    \item \textbf{BIOS} : Basic Input Output System, situ\'e sur la ROM de la carte m\`ere % Legacy, limité
    \item \textbf{UEFI} : Unified Extensible Firmware Interface, remplace le BIOS % Moderne, extensible
    \begin{itemize}
        \item Interface graphique \`a fen\^etre % Plus convivial
        \item Acc\`es \`a Internet % Mises à jour firmware
        \item Support GPT (partitions $>$ 2TB) % Nécessaire pour gros disques
        \item Architecture modulaire % Extensible par modules
    \end{itemize}
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Chargeur de démarrage (BootLoader)
% ------------------------------------------------------------
% GRUB2 est le bootloader standard Linux. Permet choisir OS/kernel au démarrage.
% Ne jamais éditer grub.cfg directement : régénéré automatiquement.

\subsection{Chargeur de d\'emarrage (BootLoader)}
% GRUB2 : charge kernel et initramfs en mémoire
% Menu au démarrage : choix OS/kernel, options de boot
% Configuration : modifier /etc/default/grub puis grub2-mkconfig
\begin{itemize}
    \item \textbf{GRUB} : GRand Unified Bootloader (version 2) % Standard Linux
    \item Fichiers GRUB2 :
    \begin{itemize}
        \item /etc/default/grub : param\`etres du menu % Timeout, options kernel
        \item /etc/grub.d/ : scripts de cr\'eation du menu % Scripts génération menu
        \item /boot/grub2/grub.cfg : fichier de configuration final (g\'en\'er\'e automatiquement) % Ne pas éditer !
    \end{itemize}
    \item Commande : \texttt{grub2-mkconfig} pour r\'eg\'en\'erer grub.cfg % Après modification /etc/default/grub
\end{itemize}

% ------------------------------------------------------------
% Sous-section : InitRamFs (Initial RAM Disk)
% ------------------------------------------------------------
% Initramfs contient modules kernel nécessaires pour accéder au disque
% racine. Si disque racine sur LVM/RAID/encrypté, modules dans initramfs.

\subsection{InitRamFs (Initial RAM Disk)}
% Initramfs : système de fichiers temporaire en RAM
% Contient modules kernel pour accéder disque racine
% Nécessaire si racine sur LVM, RAID, ou système de fichiers spécial
% Après montage racine, initramfs peut être libéré
\begin{itemize}
    \item Syst\`eme de fichiers racine temporaire % En RAM, pas sur disque
    \item Contient les pilotes n\'ecessaires pour acc\'eder au syst\`eme de fichiers racine % Modules kernel
    \item Charg\'e en m\'emoire par le bootloader % GRUB charge initramfs
    \item Permet de monter le "vrai" syst\`eme de fichiers racine % Transition vers système réel
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Gestionnaires de services
% ------------------------------------------------------------
% Deux systèmes principaux : SysVinit (ancien) et systemd (moderne).
% systemd est maintenant standard sur la plupart des distributions.

\subsection{Gestionnaires de services}

% ------------------------------------------------------------
% Sous-sous-section : SysVinit
% ------------------------------------------------------------
% Ancien système, encore présent sur certaines distributions.
% Runlevels : différents modes de fonctionnement système.

\subsubsection{SysVinit}
\begin{itemize}
    \item Processus /sbin/init (PID = 1)
    \item Niveaux d'ex\'ecution (runlevels) :
    \begin{itemize}
        \item 0 : arr\^et
        \item 1 : mode utilisateur unique (maintenance)
        \item 3 : mode multi-utilisateurs complet (texte)
        \item 5 : mode multi-utilisateurs graphique
        \item 6 : red\'emarrage
    \end{itemize}
    \item Fichier /etc/inittab : configuration
    \item R\'epertoires /etc/rc.d/rcX.d : scripts par niveau
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Systemd
% ------------------------------------------------------------
% Systemd est le gestionnaire moderne, standard sur RHEL 7+, Ubuntu 15.04+.
% Gère services, montages, sockets, timers. Plus rapide que SysVinit.

\subsubsection{Systemd}
% Systemd : gestionnaire système moderne
% Unités : services, targets, mounts, sockets, timers
% Dépendances : démarrage ordonné selon dépendances
% Journal : logs intégrés (journalctl)
\begin{itemize}
    \item Remplace /sbin/init % Processus PID 1
    \item Notion d'unit\'e (service, target, mount, socket) % Types d'unités
    \item R\'epertoires :
    \begin{itemize}
        \item /usr/lib/systemd/system : unit\'es syst\`eme par d\'efaut % Fournies par packages
        \item /etc/systemd/system : unit\'es personnalis\'ees % Priorité, surcharge
    \end{itemize}
    \item Commandes principales :
    \begin{itemize}
        \item \texttt{systemctl status service} : \'etat d'un service % État détaillé
        \item \texttt{systemctl start/stop/restart service} : contr\^ole % Contrôle service
        \item \texttt{systemctl enable/disable service} : activation au boot % Démarrage automatique
        \item \texttt{systemctl mask/unmask service} : bloquer/d\'ebloquer % Empêcher démarrage
    \end{itemize}
    \item Cibles (targets) remplacent les runlevels :
    \begin{itemize}
        \item poweroff.target (runlevel 0) % Arrêt système
        \item rescue.target (runlevel 1) % Mode maintenance
        \item multi-user.target (runlevel 3) % Multi-utilisateurs texte
        \item graphical.target (runlevel 5) % Multi-utilisateurs graphique
        \item reboot.target (runlevel 6) % Redémarrage
    \end{itemize}
\end{itemize}

\subsection{Arr\^et du syst\`eme}
\begin{itemize}
    \item \texttt{systemctl poweroff} : arr\^et et hors-tension
    \item \texttt{systemctl halt} : arr\^et sans hors-tension
    \item \texttt{systemctl reboot} : red\'emarrage
    \item \texttt{systemctl suspend} : suspension (RAM)
    \item \texttt{systemctl hibernate} : hibernation (disque)
    \item \texttt{shutdown [OPTION] [HEURE] [MESSAGE]} : arr\^et programm\'e
\end{itemize}

\subsection{Gestion des services}
\begin{itemize}
    \item \texttt{systemctl status service} : \'etat d\'etaill\'e
    \item \texttt{systemctl is-active service} : v\'erifier si actif
    \item \texttt{systemctl is-enabled service} : v\'erifier activation au boot
    \item \texttt{systemctl daemon-reload} : recharger configuration
    \item \texttt{journalctl -u service -f} : logs en temps r\'eel
\end{itemize}

\subsection{D\'epannage et journalisation}
\begin{itemize}
    \item \texttt{journalctl} : afficher tous les logs
    \item \texttt{journalctl -f} : suivre les nouveaux logs
    \item \texttt{journalctl -b} : logs du d\'emarrage actuel
    \item \texttt{journalctl -b -1} : logs du d\'emarrage pr\'ec\'edent
    \item \texttt{journalctl -u service} : logs d'un service
    \item \texttt{journalctl -p err} : logs d'erreur et sup\'erieurs
    \item \texttt{journalctl --since "1 hour ago"} : logs depuis 1 heure
\end{itemize}

% ============================================================
% SECTION 4 : GESTION DES PAQUETAGES
% ============================================================
% La gestion des paquets est fondamentale pour l'administration.
% RPM et YUM/DNF sont les outils standards sur RedHat/Fedora.
% Comprendre dépendances et dépôts est essentiel.

\section{Gestion des Paquetages}

% ------------------------------------------------------------
% Sous-section : Notion de paquetage
% ------------------------------------------------------------
% Paquet = archive + métadonnées + scripts installation.
% Gestion automatique des dépendances évite problèmes manuels.

\subsection{Notion de paquetage}
Un paquet est un fichier archive contenant le logiciel \`a installer et des r\`egles d'installation :
\begin{itemize}
    \item Gestion des d\'ependances
    \item Pr\'e-installation et post-installation
    \item Informations compl\'ementaires (nom, version, description)
\end{itemize}

\subsection{Formats de paquetages}
\begin{itemize}
    \item \textbf{tar} : format ancien, sources et binaires
    \item \textbf{rpm} : RedHat Package Manager (RedHat, SUSE, Fedora)
    \item \textbf{deb} : Debian Package (Debian, Ubuntu, Mint)
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Paquetage RPM
% ------------------------------------------------------------
% RPM : format binaire RedHat. Outil de bas niveau.
% YUM/DNF gèrent dépendances automatiquement, RPM non.

\subsection{Paquetage RPM}
% RPM : outil bas niveau, gestion manuelle dépendances
% Base de données : métadonnées packages installés
% Options : -i (install), -e (erase), -U (upgrade), -q (query), -V (verify)
\begin{itemize}
    \item Outil RPM : Redhat Package Manager % Format RedHat/Fedora/SUSE
    \item Base de donn\'ees : /var/lib/rpm/ % Métadonnées packages installés
    \item Commandes principales :
    \begin{itemize}
        \item \texttt{rpm -i fichier\_paquetage} : installer % Install package
        \item \texttt{rpm -e nom\_du\_paquetage} : d\'esinstaller % Erase/remove
        \item \texttt{rpm -U fichier\_paquetage} : mettre \`a jour % Upgrade
        \item \texttt{rpm -qa} : lister tous les packages install\'es % Query all
        \item \texttt{rpm -q package} : version install\'ee % Query version
        \item \texttt{rpm -qi package} : informations sur un package % Query info
        \item \texttt{rpm -ql package} : liste des fichiers install\'es % Query list
        \item \texttt{rpm -qf nom\_du\_fichier} : package auquel appartient un fichier % Query file
        \item \texttt{rpm -V package} : v\'erification % Verify integrity
        \item \texttt{rpm -K package.rpm} : v\'erifier signature % Check signature
    \end{itemize}
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Gestionnaire de paquets YUM
% ------------------------------------------------------------
% YUM résout automatiquement dépendances. Plus simple que RPM direct.
% Utilise dépôts configurés dans /etc/yum.repos.d/

\subsection{Gestionnaire de paquets YUM}
% YUM : résolution automatique dépendances
% Dépôts : sources packages (local, réseau)
% Cache : /var/cache/yum/ (peut être nettoyé)
Yellowdog Updater Modifier - surcouche de RPM g\'erant t\'el\'echargements et d\'ependances.

% Commandes YUM essentielles pour examen :
Commandes principales :
\begin{itemize}
    \item \texttt{yum list all} : lister tous les paquetages disponibles
    \item \texttt{yum list installed} : lister les paquetages install\'es
    \item \texttt{yum info package} : informations sur un paquetage
    \item \texttt{yum search mot\_cl\'e} : rechercher des paquetages
    \item \texttt{yum install package} : installer
    \item \texttt{yum check-update} : mises \`a jour disponibles
    \item \texttt{yum update [package]} : mettre \`a jour
    \item \texttt{yum remove package} : d\'esinstaller
    \item \texttt{yum provides fichier} : chercher un paquet contenant un fichier
    \item \texttt{yum clean all} : vider le cache
    \item \texttt{yum makecache} : mettre \`a jour la liste des packages
    \item \texttt{yum grouplist} : lister les groupes de packages
    \item \texttt{yum groupinstall "nom\_groupe"} : installer un groupe
    \item \texttt{yum history} : historique des transactions
\end{itemize}

% Comprendre fichiers /etc/passwd, /etc/shadow, /etc/group est essentiel.

\section{Gestion des Utilisateurs}

% ------------------------------------------------------------
% Sous-section : Types de comptes utilisateurs
% ------------------------------------------------------------
% UID 0 = root (super-utilisateur). UID < 1000 = comptes système.
% UID >= 1000 = utilisateurs normaux.

\subsection{Types de comptes utilisateurs}
\begin{itemize}
    \item \textbf{Super-utilisateur (root)} : UID = 0, tous les privil\`eges
    \item \textbf{Comptes syst\`emes} : UID $<$ 1000, services du syst\`eme
    \item \textbf{Comptes ordinaires} : UID $\geq$ 1000, utilisateurs normaux
\end{itemize}

\subsection{Fichiers d'enregistrement}
\begin{itemize}
    \item \textbf{/etc/passwd} : liste des utilisateurs avec descriptions
    \item \textbf{/etc/shadow} : mots de passe crypt\'es + dur\'ees de validit\'e
    \item \textbf{/etc/group} : d\'efinitions des groupes
    \item \textbf{/etc/gshadow} : mots de passe des groupes (rarement utilis\'e)
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Structure de /etc/passwd
% ------------------------------------------------------------
% Fichier texte lisible par tous. Mots de passe dans /etc/shadow (sécurité).
% Format : 7 champs séparés par ':'

\subsection{Structure de /etc/passwd}
% Format : 7 champs séparés par ':'
% Lecture publique (permissions 644), mais mots de passe dans shadow
Format : \texttt{login:passwd:uid:gid:comment:home:shell}
\begin{itemize}
    \item login : nom de connexion % Identifiant utilisateur
    \item passwd : x (indique pr\'esence de /etc/shadow) % 'x' = password dans shadow
    \item uid : num\'ero unique de l'utilisateur % User ID, 0 = root
    \item gid : num\'ero de groupe primaire % Group ID primaire
    \item comment : nom complet % GECOS field, nom réel
    \item home : r\'epertoire personnel % Home directory
    \item shell : interpr\'eteur de commandes % Shell par défaut (/bin/bash, /sbin/nologin)
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Structure de /etc/shadow
% ------------------------------------------------------------
% Fichier sécurisé : lecture root uniquement (permissions 000 ou 400).
% Contient mots de passe hashés et politiques d'expiration.

\subsection{Structure de /etc/shadow}
% Format : 9 champs séparés par ':'
% Sécurité : permissions 000 ou 400, root uniquement
% Hash : algorithmes (MD5, SHA-256, SHA-512, bcrypt)
Format : \texttt{login:passwd:last:may:must:warn:inactif:disable}
\begin{itemize}
    \item passwd : mot de passe encod\'e % Hash cryptographique
    \item last : jours depuis modification (depuis 1/1/1970) % Epoch days
    \item may : jours minimum avant changement % Minimum days
    \item must : jours maximum avant expiration % Maximum days
    \item warn : jours d'avertissement avant expiration % Warning period
    \item inactif : p\'eriode d'inactivit\'e maximale % Inactive days
    \item disable : date de fermeture du compte % Account expiration
\end{itemize}

\subsection{Commandes de gestion des utilisateurs}

\subsubsection{Cr\'eation}
\begin{itemize}
    \item \texttt{useradd [options] nom\_utilisateur}
    \begin{itemize}
        \item \texttt{-u} : sp\'ecifier UID
        \item \texttt{-c} : commentaire (nom complet)
        \item \texttt{-d} : r\'epertoire personnel
        \item \texttt{-e} : date d'expiration (AAAA-MM-JJ)
        \item \texttt{-g} : groupe primaire
        \item \texttt{-G} : groupes secondaires
        \item \texttt{-m} : cr\'eer le r\'epertoire personnel
        \item \texttt{-s} : shell
        \item \texttt{-r} : cr\'eer un compte syst\`eme
    \end{itemize}
    \item \texttt{passwd utilisateur} : d\'efinir le mot de passe
\end{itemize}

\subsubsection{Modification}
\begin{itemize}
    \item \texttt{usermod [options] utilisateur}
    \begin{itemize}
        \item \texttt{-s} : changer le shell
        \item \texttt{-c} : modifier le commentaire
        \item \texttt{-aG} : ajouter \`a des groupes (sans \'ecraser)
        \item \texttt{-d -m} : d\'eplacer le home
        \item \texttt{-L} : verrouiller le compte
        \item \texttt{-U} : d\'everrouiller
        \item \texttt{-e} : expirer le compte
    \end{itemize}
\end{itemize}

\subsubsection{Suppression}
\begin{itemize}
    \item \texttt{userdel utilisateur} : supprimer (garde le home)
    \item \texttt{userdel -r utilisateur} : supprimer avec le home
    \item \texttt{userdel -rf utilisateur} : forcer m\^eme si connect\'e
\end{itemize}

\subsubsection{Gestion des mots de passe}
\begin{itemize}
    \item \texttt{passwd [options] utilisateur}
    \begin{itemize}
        \item \texttt{-d} : supprimer le mot de passe
        \item \texttt{-l} : verrouiller
        \item \texttt{-u} : d\'everrouiller
        \item \texttt{-e} : faire expirer
        \item \texttt{-n jours} : jours minimum avant changement
        \item \texttt{-x jours} : jours maximum avant expiration
        \item \texttt{-w jours} : d\'elai d'avertissement
        \item \texttt{-i jours} : d\'elai avant d\'esactivation
    \end{itemize}
    \item \texttt{chage [options] utilisateur} : modifier validit\'e du mot de passe
\end{itemize}

\subsection{Gestion des groupes}

\subsubsection{Cr\'eation et modification}
\begin{itemize}

    \begin{itemize}
        \item \texttt{-g GID} : sp\'ecifier GID
        \item \texttt{-r} : cr\'eer un groupe syst\`eme
    \end{itemize}
    \item \texttt{groupmod [options] nom\_groupe}
    \begin{itemize}
        \item \texttt{-g GID} : changer GID

\subsubsection{Gestion des membres}
\begin{itemize}

% ------------------------------------------------------------
% Sous-section : Droits d'accès
% ------------------------------------------------------------
% Permissions Unix : base de la sécurité. Trois catégories (u, g, o)
% et trois droits (r, w, x). Calcul octal : r=4, w=2, x=1.

\subsection{Droits d'acc\`es}

% ------------------------------------------------------------
% Sous-sous-section : Permissions traditionnelles
% ------------------------------------------------------------
% Permissions : propriétaire (u), groupe (g), autres (o)
% Droits : lecture (r=4), écriture (w=2), exécution (x=1)
% Mode octal : somme des droits (ex: 755 = rwxr-xr-x)

\subsubsection{Permissions traditionnelles}
Trois cat\'egories : propri\'etaire (u), groupe (g), autres (o)
\begin{itemize}
    \item \textbf{r} (read) : lecture
    \item \textbf{w} (write) : \'ecriture
    \item \textbf{x} (execute) : ex\'ecution
\end{itemize}

\subsubsection{Modification des permissions}
\begin{itemize}
    \item \texttt{chmod [mode] fichier}
    \begin{itemize}
        \item Mode symbolique : \texttt{u+x}, \texttt{g-rw}, \texttt{o=r}, \texttt{a+r}
        \item Mode absolu : \texttt{755}, \texttt{644}, \texttt{600}, \texttt{777}
    \end{itemize}
    \item \texttt{chmod -R mode r\'epertoire} : r\'ecursif
\end{itemize}

\subsubsection{Changement de propri\'etaire}
\begin{itemize}
    \item \texttt{chown utilisateur[:groupe] fichier} : changer propri\'etaire
    \item \texttt{chgrp groupe fichier} : changer groupe
    \item \texttt{chown -R utilisateur:groupe r\'epertoire} : r\'ecursif
\end{itemize}

% Sous-section : Gestion des privilèges : sudo
% ------------------------------------------------------------
% Sudo permet exécuter commandes avec privilèges élevés sans être root.
% Configuration dans /etc/sudoers. Toujours utiliser visudo (vérification syntaxe).

\subsection{Gestion des privil\`eges : sudo}
% Sudo : "substitute user do" ou "super user do"
% Permet élévation privilèges temporaire
% visudo : éditeur sécurisé (vérifie syntaxe avant sauvegarde)
% Format : qui où=(qui_cible) quoi
\begin{itemize}
    \item Fichier de configuration : /etc/sudoers % Configuration sudo
    \item \texttt{visudo} : \'editer de fa\c{c}on s\'ecuris\'ee % Toujours utiliser visudo !
    \item Format : \texttt{utilisateur h\^ote=(utilisateur\_cible) commandes} % Syntaxe sudoers
    \item Exemples :
    \begin{itemize}
        \item \texttt{jean ALL=(ALL:ALL) ALL} : tout faire % Privilèges complets
        \item \texttt{\%wheel ALL=(ALL) ALL} : groupe wheel % Groupe avec privilèges
        \item \texttt{marie ALL=(root) NOPASSWD:/usr/sbin/reboot} : sans mot de passe % Commande spécifique sans password
    \end{itemize}
\end{itemize}

% SECTION 6 : GESTION DES DISQUES ET SYSTÈMES DE FICHIERS
% ============================================================

% Comprendre MBR vs GPT, types de systèmes de fichiers, inodes.

\section{Gestion des Disques et Syst\`emes de Fichiers}

% ------------------------------------------------------------
% Sous-section : Types de disques
% ------------------------------------------------------------
% Différents types de stockage : HDD (mécanique), SSD (flash), NVMe (PCIe).
% Performances et coûts varient.

\subsection{Types de disques}
\begin{itemize}
    \item \textbf{HDD} : Disque dur m\'ecanique, \'economique, lent
    \item \textbf{SSD} : Disque \'electronique, rapide, silencieux
    \item \textbf{NVMe} : Tr\`es haute performance
\end{itemize}

\subsection{Partitionnement}

% ------------------------------------------------------------
% Sous-sous-section : MBR vs GPT
% ------------------------------------------------------------
% MBR : ancien standard, limitations. GPT : moderne, plus flexible.
% GPT recommandé pour disques > 2TB et systèmes UEFI.

\subsubsection{MBR vs GPT}
% MBR : Master Boot Record, secteur 0 du disque
% GPT : GUID Partition Table, plus moderne, redondance
% Conversion possible : gdisk peut convertir MBR->GPT
\begin{itemize}
    \item \textbf{MBR} : Master Boot Record
    \begin{itemize}
        \item Limite : 4 partitions primaires maximum % Ou 3 primaires + 1 étendue
        \item Taille max : 2.2 To par partition % Limitation 32 bits
        \item Compatible BIOS % Legacy
    \end{itemize}
    \item \textbf{GPT} : GUID Partition Table
    \begin{itemize}
        \item Jusqu'\`a 128 partitions % Standard, peut être plus
        \item Taille tr\`es grande (jusqu'\`a 9.4 Zo) % 64 bits, énorme
        \item Compatible UEFI et BIOS % Moderne, rétrocompatible
    \end{itemize}
\end{itemize}

\subsubsection{Organisation des disques sous Linux}
\begin{itemize}
    \item Format : /dev/sdXY ou /dev/nvme0n1pX
    \item Types de bus : hd (IDE), sd (SATA/SCSI/SSD), nvme (NVMe)
    \item UUID : identifiant unique universel
\end{itemize}

\subsubsection{Commandes d'information}
\begin{itemize}
    \item \texttt{lsblk} : afficher informations sur disques et partitions
    \item \texttt{blkid} : obtenir UUID
    \item \texttt{df -h} : taux de remplissage
    \item \texttt{du -h [chemin]} : espace occup\'e par une arborescence
\end{itemize}

\subsubsection{Cr\'eation de partitions}
\begin{itemize}
    \item \texttt{fdisk /dev/sdX} : manipuler table de partitions (interactif)
    \item \texttt{gdisk /dev/sdX} : pour GPT
    \item \texttt{parted} : mode interactif, script ou mixte
    \item Commandes fdisk : n (nouvelle), d (supprimer), t (type), w (sauvegarder), g (GPT)
\end{itemize}

\subsection{Syst\`emes de fichiers}

\subsubsection{D\'efinition}
Un syst\`eme de fichiers d\'efinit l'organisation d'un volume physique ou logique, permettant de stocker et organiser les informations dans des fichiers.

\subsubsection{Syst\`emes de fichiers GNU/Linux}
\begin{itemize}
    \item \textbf{ext2} : stable et robuste
    \item \textbf{ext3} : + journalisation
    \item \textbf{ext4} : volumes $>$ 1To, timestamps nanosec
    \item \textbf{XFS} : performance fichiers volumineux
    \item \textbf{Btrfs} : snapshots et compression
    \item \textbf{ZFS} : int\'egration stockage avanc\'ee
\end{itemize}

\subsubsection{Initialisation (formatage)}
\begin{itemize}
    \item \texttt{mkfs -t type partition}
    \item \texttt{mkfs.ext4 /dev/sda1}
    \item \texttt{lsblk -f} : v\'erifier le type
\end{itemize}

\subsection{Repr\'esentation des fichiers sur disque}

\subsubsection{Concepts}
\begin{itemize}
    \item \textbf{Bloc} : unit\'e de transfert entre disque et m\'emoire (souvent 4096 octets)
    \item \textbf{Inode} : descripteur d'un fichier contenant les m\'etadonn\'ees
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Inode
% ------------------------------------------------------------
% Inode = métadonnées fichier. Nom fichier dans répertoire parent.
% Limite inodes par partition : vérifier avec df -i.

\subsubsection{Inode}
% Inode : index node, descripteur fichier
% Chaque fichier a un inode unique dans le système de fichiers
% Nom fichier : stocké dans répertoire parent, pointe vers inode
% Limite : nombre max inodes défini à la création du système de fichiers
Contient :
\begin{itemize}
    \item Type de l'inode (fichier, r\'epertoire) % Type : fichier, répertoire, lien, device
    \item Propri\'etaire (UID, GID) % Propriétaire et groupe
    \item Droits d'acc\`es % Permissions (rwx)
    \item Dates (atime, mtime, ctime) % Access, modify, change time
    \item Taille % Taille en octets
    \item Adresse : pointeurs sur les blocs du disque % Pointeurs vers blocs données
\end{itemize}
\textbf{Important} : Les inodes ne contiennent pas les noms des fichiers. % Nom dans répertoire parent

% ------------------------------------------------------------
% Sous-sous-section : Liens
% ------------------------------------------------------------
% Deux types : liens physiques (hard links) et symboliques (soft links).
% Liens physiques : même inode, même fichier. Symboliques : pointeur.

\subsubsection{Liens}
% Hard link : même inode, compteur de références
% Soft link : fichier séparé pointant vers nom de fichier
% Suppression : hard link décrémente compteur, soft link devient orphelin
\begin{itemize}
    \item \textbf{Lien physique} : \texttt{ln fichier lien}
    \begin{itemize}
        \item Partage le m\^eme inode % Même inode = même fichier
        \item Reste valide si fichier d\'eplac\'e % Tant que inode existe
        \item Doit rester dans la m\^eme partition % Limitation système de fichiers
    \end{itemize}
    \item \textbf{Lien symbolique} : \texttt{ln -s fichier lien}
    \begin{itemize}
        \item Pointe vers le nom du fichier % Contient chemin, pas inode
        \item Peut traverser les partitions % Pas de limitation
        \item Ressemble aux raccourcis % Peut pointer vers inexistant
    \end{itemize}
\end{itemize}

\subsection{Montage d'une partition}

\subsubsection{Commandes}
\begin{itemize}
    \item \texttt{mount partition point\_de\_montage} : monter
    \item \texttt{mount -t type partition point} : sp\'ecifier type
    \item \texttt{mount -a} : monter tous les SF de /etc/fstab
    \item \texttt{umount point\_de\_montage} : d\'emonter
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Fichier /etc/fstab
% ------------------------------------------------------------
% /etc/fstab : montage automatique au démarrage.
% UUID recommandé plutôt que /dev/sdX (plus stable).

\subsubsection{Fichier /etc/fstab}
% fstab : File System Table
% Montage automatique au boot (mount -a)
% UUID : identifiant unique, plus fiable que /dev/sdX
% Options : defaults, noauto, user, etc.
Configuration statique des montages. Structure :
\begin{verbatim}
périphérique  point_montage  type  options  dump  fsck
\end{verbatim}
\begin{itemize}
    \item dump : activer sauvegarde (1 ou 0) % Utilisé par dump (backup)
    \item fsck : ordre de v\'erification (0, 1, 2, 3) % 0=pas de vérif, 1=racine, 2+=autres
\end{itemize}

\subsection{Gestion de l'espace Swap}

% ------------------------------------------------------------
% Sous-sous-section : Définition du Swap
% ------------------------------------------------------------
% Swap : extension mémoire sur disque. Plus lent que RAM.
% Taille recommandée : 1-2x RAM pour systèmes avec peu de RAM.

\subsubsection{D\'efinition}
% Swap : espace disque utilisé comme mémoire virtuelle
% Utilisé quand RAM saturée : pages inactives déplacées
% Performance : beaucoup plus lent que RAM (disque vs mémoire)
Zone de swap utilis\'ee lorsque la m\'emoire physique (RAM) est remplie. Les pages m\'emoire inactives sont d\'eplac\'ees dans la zone de swap.

\subsubsection{Configuration}
\begin{enumerate}
    \item Cr\'eer un fichier swap : \texttt{fallocate -l 1G /swapfile} ou \texttt{dd if=/dev/zero of=/swapfile bs=1M count=1024}
    \item Verrouiller permissions : \texttt{chmod 600 /swapfile}
    \item Marquer comme swap : \texttt{mkswap /swapfile}
    \item Activer : \texttt{swapon /swapfile}
    \item Rendre permanent : ajouter dans /etc/fstab
\end{enumerate}

\subsubsection{Commandes}
\begin{itemize}
    \item \texttt{swapon --show} : afficher swaps actifs
    \item \texttt{free -h} : \'etat m\'emoire et swap
    \item \texttt{swapoff /swapfile} : d\'esactiver
\end{itemize}

% SECTION 7 : SÉCURITÉ SOUS LINUX
% ============================================================
% Sécurité : ACLs, droits spéciaux (setuid, setgid, sticky),
% SELinux. Comprendre ces mécanismes est crucial pour l'examen.

\section{S\'ecurit\'e sous Linux}

% ------------------------------------------------------------
% Sous-section : ACLs (Liste de Contrôle d'Accès)
% ------------------------------------------------------------
% ACLs : permissions avancées au-delà des 3 catégories standard.
% Permet permissions spécifiques par utilisateur/groupe.

\subsection{ACLs : Liste de Contr\^ole d'Acc\`es}

% ------------------------------------------------------------
% Sous-sous-section : Définition des ACLs
% ------------------------------------------------------------
% ACLs : permissions granulaire. Plusieurs utilisateurs/groups
% peuvent avoir permissions différentes sur même fichier.

\subsubsection{D\'efinition}
% ACLs : Access Control Lists
% Extension permissions Unix classiques (u, g, o)
% Permet permissions spécifiques par utilisateur/groupe
% Nécessite système de fichiers avec support ACL (ext4, XFS)
Les ACLs permettent d'\'etendre le nombre d'utilisateurs et de groupes ayant des droits sur un m\^eme fichier, au-del\`a des permissions traditionnelles.

\subsubsection{Commandes}
\begin{itemize}
    \item \texttt{getfacl fichier} : consulter les ACLs
    \item \texttt{setfacl -m u:utilisateur:rw fichier} : modifier ACLs
    \item \texttt{setfacl -m g:groupe:r fichier} : droits pour un groupe
    \item \texttt{setfacl -m o::- fichier} : refuser aux autres
    \item \texttt{setfacl -R -m u:utilisateur:rw r\'epertoire} : r\'ecursif
    \item \texttt{setfacl -b fichier} : supprimer toutes les ACLs
    \item \texttt{setfacl -x u:utilisateur fichier} : retirer permissions
\end{itemize}

\subsubsection{ACLs par d\'efaut}
\begin{itemize}
    \item \texttt{setfacl -m d:o::- r\'epertoire} : ACLs par d\'efaut
    \item S'appliquent aux nouveaux fichiers/r\'epertoires cr\'e\'es
    \item \texttt{setfacl -k r\'epertoire} : annuler ACLs par d\'efaut
\end{itemize}

\subsubsection{Masque}
Le masque limite les autorisations effectives pour les groupes et entr\'ees sp\'ecifiques.
\begin{itemize}
    \item \texttt{setfacl -m m::rwx fichier} : d\'efinir un masque
\end{itemize}

\subsection{Droits sp\'eciaux}

% ------------------------------------------------------------
% Sous-sous-section : Setuid
% ------------------------------------------------------------
% Setuid : exécution avec droits propriétaire. Risque sécurité
% si mal utilisé. Exemple : /usr/bin/passwd (modifie /etc/shadow).

\subsubsection{Setuid}
% Setuid : Set User ID
% Permet élévation privilèges temporaire
% Risque sécurité : programme compromis = compromission système
% Exemples : passwd, su, sudo
Permet \`a un utilisateur non-root d'ex\'ecuter un programme avec les droits du propri\'etaire du programme.
\begin{itemize}
    \item \texttt{chmod u+s programme} : activer % Setuid bit
    \item \texttt{chmod 4xxx programme} : mode absolu % 4xxx = setuid activé
    \item Affichage : 's' \`a la place de 'x' pour user % 's' = setuid + execute
    \item 'S' si pas de droit x % 'S' = setuid sans execute
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Setgid
% ------------------------------------------------------------
% Setgid : sur fichiers = exécution avec droits groupe.
% Sur répertoires = héritage groupe (utile pour collaboration).

\subsubsection{Setgid}
% Setgid : Set Group ID
% Sur fichiers : exécution avec privilèges groupe propriétaire
% Sur répertoires : héritage groupe pour nouveaux fichiers
% Utile : collaboration équipe, fichiers créés appartiennent au groupe
\begin{itemize}
    \item \textbf{Sur fichiers} : s'ex\'ecute avec les privil\`eges du groupe propri\'etaire % Exécution groupe
    \item \textbf{Sur r\'epertoires} : fichiers cr\'e\'es h\'eritent du groupe du r\'epertoire % Héritage groupe
    \item \texttt{chmod g+s r\'epertoire} : activer % Setgid bit
    \item \texttt{chmod 2xxx r\'epertoire} : mode absolu % 2xxx = setgid activé
    \item Affichage : 's' \`a la place de 'x' pour groupe % 's' = setgid + execute
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Sticky Bit
% ------------------------------------------------------------
% Sticky bit : sur répertoires partagés (ex: /tmp).
% Empêche suppression fichiers d'autres utilisateurs.

\subsubsection{Sticky Bit}
% Sticky bit : sur répertoires uniquement
% Empêche suppression fichiers par non-propriétaires
% Exemple classique : /tmp (tous peuvent créer, seul propriétaire peut supprimer)
% Historique : sur fichiers exécutables, gardait en mémoire (obsolète)
Sur les r\'epertoires, emp\^eche les utilisateurs non-root de supprimer ou modifier les fichiers dont ils ne sont pas propri\'etaires.
\begin{itemize}
    \item \texttt{chmod +t r\'epertoire} : activer % Sticky bit
    \item \texttt{chmod 1xxx r\'epertoire} : mode absolu % 1xxx = sticky activé
    \item Affichage : 't' \`a la place de 'x' pour other % 't' = sticky + execute
    \item 'T' si pas de droit x % 'T' = sticky sans execute
\end{itemize}

% ------------------------------------------------------------
% Sous-section : SELinux (Security-Enhanced Linux)
% ------------------------------------------------------------
% SELinux : sécurité renforcée, contrôle d'accès obligatoire (MAC).
% Même root doit respecter politiques. Mode enforcing/permissive/disabled.

\subsection{SELinux : Security-Enhanced Linux}

% ------------------------------------------------------------
% Sous-sous-section : Introduction à SELinux
% ------------------------------------------------------------
% SELinux : couche sécurité supplémentaire au-dessus permissions Unix.
% Contrôle d'accès obligatoire (MAC) vs discrétionnaire (DAC).

\subsubsection{Introduction}
% SELinux : développé par NSA, intégré dans noyau Linux
% Couche supplémentaire : même si permissions Unix OK, SELinux peut bloquer
% Politique : règles définissant qui peut faire quoi
SELinux est une architecture de s\'ecurit\'e pour syst\`emes Linux permettant aux administrateurs de mieux contr\^oler les acc\`es au syst\`eme. C'est une couche suppl\'ementaire de s\'ecurit\'e.

% ------------------------------------------------------------
% Sous-sous-section : DAC vs MAC
% ------------------------------------------------------------
% DAC : permissions classiques Unix (propriétaire décide).
% MAC : politique système obligatoire (même root respecte règles).

\subsubsection{DAC vs MAC}
% DAC : Discretionary = discrétionnaire, propriétaire contrôle
% MAC : Mandatory = obligatoire, politique système
% SELinux = implémentation MAC avec Type Enforcement
\begin{itemize}
    \item \textbf{DAC} (Discretionary Access Control) : contr\^ole d'acc\`es discr\'etionnaire
    \begin{itemize}
        \item Permissions classiques (r, w, x) % Propriétaire décide
        \item Droits sp\'eciaux (setuid, setgid, sticky bit) % Extensions permissions
        \item ACLs % Permissions avancées
    \end{itemize}
    \item \textbf{MAC} (Mandatory Access Control) : contr\^ole d'acc\`es obligatoire
    \begin{itemize}
        \item Politique de s\'ecurit\'e \`a laquelle m\^eme root doit se soumettre % Même root limité
        \item Impl\'ementation : SELinux avec mod\`ele TE (Type Enforcement) % Types et règles
    \end{itemize}
\end{itemize}

\subsubsection{Concepts SELinux}
\begin{itemize}
    \item \textbf{Sujet} : entit\'e effectuant une action (processus, service, utilisateur)
    \item \textbf{Objet} : ressource sur laquelle une action est r\'ealis\'ee (fichier, r\'epertoire, socket, port)
    \item \textbf{R\`egles} : ensembles de r\`egles contr\^olant quelles actions un sujet peut effectuer sur un objet
    \item \textbf{Labeling} : chaque ressource poss\`ede une \'etiquette (contexte SELinux)
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Contexte SELinux
% ------------------------------------------------------------
% Contexte = étiquette SELinux sur chaque ressource.
% Format : user:role:type:level. Type est le plus important.

\subsubsection{Contexte SELinux}
% Contexte : étiquette SELinux sur fichiers, processus, ports
% Format : 4 composants séparés par ':'
% Type : composant principal, détermine permissions
Format : \texttt{user:role:type:level}
\begin{itemize}
    \item \textbf{user} : utilisateur SELinux (system\_u, unconfined\_u) % Identité SELinux
    \item \textbf{role} : r\^ole (object\_r pour objets) % Rôle dans politique
    \item \textbf{type} : le plus important, d\'etermine les permissions % Type = permissions
    \begin{itemize}
        \item httpd\_t : processus serveur web % Type processus Apache
        \item httpd\_sys\_content\_t : fichiers accessibles par Apache % Type fichiers web
        \item shadow\_t : fichier des mots de passe % Type fichiers sensibles
    \end{itemize}
    \item \textbf{level} : niveau de sensibilit\'e (s0) % Multi-Level Security (MLS)
\end{itemize}

% ------------------------------------------------------------
% Sous-sous-section : Modes de fonctionnement SELinux
% ------------------------------------------------------------
% Trois modes : enforcing (production), permissive (débogage), disabled.
% Changer mode nécessite souvent redémarrage.

\subsubsection{Modes de fonctionnement}
% Enforcing : production, sécurité maximale
% Permissive : développement, logs sans blocage
% Disabled : désactivé (nécessite redémarrage pour réactiver)
\begin{itemize}
    \item \textbf{Enforcing} : mode strict, acc\`es restreints, refus des acc\`es non autoris\'es % Production
    \item \textbf{Permissive} : mode d\'ebogage, r\`egles v\'erifi\'ees, erreurs enregistr\'ees mais acc\`es non bloqu\'e % Débogage
    \item \textbf{Disabled} : d\'esactiv\'e, aucune restriction % Désactivé
\end{itemize}

\subsubsection{Commandes}
\begin{itemize}
    \item \texttt{getenforce} : mode actuel
    \item \texttt{setenforce enforcing} : passer en enforcing
    \item \texttt{setenforce permissive} : passer en permissif
    \item \texttt{sestatus} : \'etat g\'en\'eral
    \item \texttt{ls -Z fichier} : contexte d'un fichier
    \item \texttt{ps -Z} : contexte des processus
    \item \texttt{chcon -t type fichier} : changer contexte
    \item \texttt{restorecon fichier} : restaurer contexte par d\'efaut
    \item \texttt{semanage fcontext -l} : lister contextes par d\'efaut
\end{itemize}

\subsubsection{Configuration}
Fichier /etc/selinux/config :
\begin{verbatim}
SELINUX=enforcing|permissive|disabled
SELINUXTYPE=targeted
\end{verbatim}

\subsubsection{Exemples de types SELinux}
\begin{itemize}
    \item \texttt{httpd\_sys\_content\_t} : contenu accessible par Apache
    \item \texttt{device\_t} : fichiers associ\'es aux p\'eriph\'eriques
    \item \texttt{home\_dir\_t} : r\'epertoires personnels
    \item \texttt{var\_log\_t} : fichiers de journaux
    \item \texttt{shadow\_t} : fichiers de mots de passe
\end{itemize}

\section{Gestion des Processus}

% ------------------------------------------------------------
% Sous-section : Notions de base
% ------------------------------------------------------------
% Processus = instance d'un programme en cours d'exécution.
% Chaque processus a un PID unique. Processus parent/enfant.

\subsection{Notions de base}
\begin{itemize}
    \item \textbf{Processus} : instance d'un programme en cours d'ex\'ecution % Programme en mémoire
    \item \textbf{PID} : Process ID, identifiant unique du processus % Unique, réutilisé après terminaison
    \item \textbf{PPID} : Parent Process ID, PID du processus parent % Hiérarchie processus
    \item \textbf{Processus parent/enfant} : relation hi\'erarchique % Fork/exec modèle
    \item \textbf{Processus zombie} : processus termin\'e mais non r\'ecup\'er\'e par parent % Ressources non libérées
    \item \textbf{Processus orphelin} : processus dont le parent est mort, adopt\'e par init (PID 1) % Init devient parent
\end{itemize}

% ------------------------------------------------------------
% Sous-section : États d'un processus
% ------------------------------------------------------------
% États : Running, Sleeping, Stopped, Zombie.
% Transition entre états selon événements système.

\subsection{\'Etats d'un processus}
\begin{itemize}
    \item \textbf{R (Running)} : en cours d'ex\'ecution ou pr\^et \`a s'ex\'ecuter % Actif ou prêt
    \item \textbf{S (Sleeping)} : en attente d'un \'ev\'enement (interruptible) % Attente I/O, signal
    \item \textbf{D (Disk sleep)} : en attente d'un \'ev\'enement (non interruptible) % Attente disque, critique
    \item \textbf{T (Stopped)} : arr\^et\'e par un signal (SIGSTOP, SIGTSTP) % Suspendu, Ctrl+Z
    \item \textbf{Z (Zombie)} : termin\'e mais non r\'ecup\'er\'e par le parent % Ressources non libérées
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Commandes de visualisation
% ------------------------------------------------------------
% ps : instantané. top/htop : temps réel. pstree : arborescence.

\subsection{Commandes de visualisation}

\subsubsection{ps (Process Status)}
% ps : affiche instantané des processus
% Options : -e (tous), -f (format complet), -u (par utilisateur), -aux (BSD style)
\begin{itemize}
    \item \texttt{ps} : processus de la session courante % Session actuelle
    \item \texttt{ps -e} : tous les processus % Tous les processus
    \item \texttt{ps -f} : format complet (UID, PID, PPID, CMD) % Format détaillé
    \item \texttt{ps -u utilisateur} : processus d'un utilisateur % Filtre utilisateur
    \item \texttt{ps aux} : tous les processus (format BSD) % Style BSD, très utilisé
    \item \texttt{ps -ef} : format étendu, tous processus % Format étendu
    \item \texttt{ps -o pid,ppid,cmd} : colonnes personnalisées % Colonnes spécifiques
\end{itemize}

\subsubsection{top et htop}
% top : monitoring temps réel, interactif
% htop : version améliorée, plus conviviale
\begin{itemize}
    \item \texttt{top} : affichage dynamique des processus % Temps réel, interactif
    \begin{itemize}
        \item Tri : P (CPU), M (m\'emoire), T (temps) % Trier par colonne
        \item Actions : k (tuer), r (renice), q (quitter) % Actions interactives
    \end{itemize}
    \item \texttt{htop} : version am\'elior\'ee de top % Interface améliorée
    \item \texttt{top -u utilisateur} : filtrer par utilisateur % Filtre utilisateur
    \item \texttt{top -p PID} : suivre processus sp\'ecifique % Processus spécifique
\end{itemize}

\subsubsection{pstree}
% pstree : arborescence des processus
% Montre relations parent/enfant visuellement
\begin{itemize}
    \item \texttt{pstree} : arborescence des processus % Hiérarchie visuelle
    \item \texttt{pstree -p} : afficher les PIDs % Avec PIDs
    \item \texttt{pstree -u} : afficher les utilisateurs % Avec utilisateurs
    \item \texttt{pstree utilisateur} : processus d'un utilisateur % Filtre utilisateur
\end{itemize}

% ------------------------------------------------------------
% Sous-section : Exécution de processus
% ------------------------------------------------------------
% Exécution foreground/background. Jobs. nohup pour processus persistants.

\subsection{Ex\'ecution de processus}

\subsubsection{Foreground et Background}
% Foreground : processus bloquant terminal
% Background : processus en arrière-plan, terminal libre
\begin{itemize}
    \item \textbf{Foreground} : processus au premier plan, bloque le terminal % Bloquant
    \item \textbf{Background} : processus en arri\`ere-plan, terminal libre % Non bloquant
    \item \texttt{commande \&} : lancer en arri\`ere-plan % & à la fin
    \item \texttt{Ctrl+Z} : suspendre processus foreground (SIGTSTP) % Suspendre
    \item \texttt{bg} : reprendre en arri\`ere-plan % Background
    \item \texttt{fg} : ramener au premier plan % Foreground
    \item \texttt{fg \%1} : ramener job num\'ero 1 % Job spécifique
\end{itemize}

\subsubsection{Gestion des jobs}
% Jobs : processus lancés depuis shell actuel
% jobs : liste des jobs de la session
\begin{itemize}
    \item \texttt{jobs} : lister les jobs de la session % Jobs actifs
    \item \texttt{jobs -l} : avec PIDs % Avec PIDs
    \item \texttt{kill \%1} : tuer job num\'ero 1 % Tuer job
    \item \texttt{nohup commande \&} : processus persistant apr\`es d\'econnexion % Survit à déconnexion
    \item \texttt{disown} : d\'etacher job du shell % Détacher du shell
\end{itemize}

% Sous-section : Signaux
% ------------------------------------------------------------
% Signaux : communication inter-processus. SIGTERM, SIGKILL, SIGSTOP, etc.
% kill : envoyer signal. killall : par nom.

\subsection{Signaux}

\subsubsection{D\'efinition}
% Signaux : mécanisme de communication inter-processus
% Permet contrôler processus (arrêt, reprise, redémarrage)
Les signaux sont des m\'ecanismes de communication inter-processus permettant de contr\^oler l'ex\'ecution des processus.

\subsubsection{Signaux courants}
\begin{itemize}
    \item \textbf{SIGHUP (1)} : hang up, red\'emarrer processus % Terminaison gracieuse, rechargement config
    \item \textbf{SIGINT (2)} : interruption (Ctrl+C) % Interruption utilisateur
    \item \textbf{SIGQUIT (3)} : quitter avec core dump (Ctrl+\textbackslash) % Core dump
    \item \textbf{SIGTERM (15)} : terminaison gracieuse (d\'efaut kill) % Arrêt propre
    \item \textbf{SIGKILL (9)} : terminaison forc\'ee (non interceptable) % Force brute, non ignorable
    \item \textbf{SIGSTOP (19)} : arr\^et temporaire (non interceptable) % Suspendre
    \item \textbf{SIGCONT (18)} : continuer processus arr\^et\'e % Reprendre
    \item \textbf{SIGTSTP (20)} : arr\^et depuis terminal (Ctrl+Z) % Suspendre depuis terminal
\end{itemize}

\subsubsection{Commandes}
\begin{itemize}
    \item \texttt{kill PID} : envoyer SIGTERM (terminaison gracieuse) % Par défaut SIGTERM
    \item \texttt{kill -9 PID} : envoyer SIGKILL (terminaison forc\'ee) % Force brute
    \item \texttt{kill -HUP PID} : envoyer SIGHUP (red\'emarrage) % Rechargement
    \item \texttt{kill -STOP PID} : suspendre processus % Suspendre
    \item \texttt{kill -CONT PID} : reprendre processus % Reprendre
    \item \texttt{killall nom\_processus} : tuer tous processus par nom % Par nom
    \item \texttt{pkill nom\_processus} : tuer processus par nom (pattern) % Pattern matching
    \item \texttt{killall -9 firefox} : forcer arr\^et Firefox % Exemple pratique
\end{itemize}

\subsection{Priorit\'es et nice}
% Nice : valeur de priorité d'exécution

\begin{itemize}
    \item \textbf{Nice} : valeur de priorit\'e d'ex\'ecution % Priorité CPU
    \item Plage : -20 (priorit\'e maximale) \`a +19 (priorit\'e minimale) % Valeurs possibles
    \item Par d\'efaut : nice = 0 % Priorité normale
    \item \texttt{nice -n 10 commande} : lancer avec nice = 10 % Lancer avec priorité
    \item \texttt{renice +10 PID} : modifier priorit\'e processus existant % Modifier priorité
    \item \texttt{renice -5 -u utilisateur} : modifier pour tous processus utilisateur % Par utilisateur
    \item Seul root peut utiliser nice n\'egatif % Privilège root nécessaire
\end{itemize}

\section{T\^aches Planifi\'ees}

% ------------------------------------------------------------
% Sous-section : Cron

\subsection{Cron}

\subsubsection{D\'efinition}
% Cron : démon pour tâches périodiques
% Crontab : fichier de configuration des tâches
Cron est un d\'emon permettant d'ex\'ecuter des commandes \`a intervalles r\'eguliers.

\subsubsection{Format crontab}
% Format : 5 champs + commande
% Minute (0-59), Heure (0-23), Jour (1-31), Mois (1-12), Jour semaine (0-7, 0 et 7 = dimanche)
Format : \texttt{minute heure jour mois jour\_semaine commande}
\begin{itemize}
    \item \textbf{minute} : 0-59 % Minutes
    \item \textbf{heure} : 0-23 % Heures
    \item \textbf{jour} : 1-31 % Jour du mois
    \item \textbf{mois} : 1-12 ou jan-dec % Mois
    \item \textbf{jour\_semaine} : 0-7 (0 et 7 = dimanche) ou sun-sat % Jour semaine
    \item Caract\`eres sp\'eciaux :
    \begin{itemize}
        \item \texttt{*} : toutes les valeurs % Toutes valeurs
        \item \texttt{,} : liste de valeurs (ex: 1,3,5) % Liste
        \item \texttt{-} : plage (ex: 1-5) % Plage
        \item \texttt{/} : intervalle (ex: */5 = toutes les 5 minutes) % Intervalle
    \end{itemize}
\end{itemize}

\subsubsection{Exemples crontab}
\begin{itemize}
    \item \texttt{0 * * * *} : toutes les heures \`a la minute 0 % Chaque heure
    \item \texttt{0 0 * * *} : tous les jours \`a minuit % Quotidien
    \item \texttt{0 0 * * 0} : tous les dimanches \`a minuit % Hebdomadaire
    \item \texttt{*/5 * * * *} : toutes les 5 minutes % Toutes les 5 minutes
    \item \texttt{0 9-17 * * 1-5} : chaque heure de 9h \`a 17h, jours ouvrables % Heures ouvrables
    \item \texttt{0 0 1 * *} : premier jour de chaque mois % Mensuel
\end{itemize}

\subsubsection{Commandes crontab}
\begin{itemize}
    \item \texttt{crontab -e} : \'editer crontab utilisateur % Éditer
    \item \texttt{crontab -l} : lister crontab utilisateur % Lister
    \item \texttt{crontab -r} : supprimer crontab utilisateur % Supprimer
    \item \texttt{crontab -u utilisateur -e} : \'editer crontab d'un utilisateur (root) % Par utilisateur
    \item \texttt{crontab fichier} : installer crontab depuis fichier % Depuis fichier
\end{itemize}

\subsubsection{Fichiers et r\'epertoires}
\begin{itemize}
    \item \texttt{/var/spool/cron/} : crontabs utilisateurs % Crontabs utilisateurs
    \item \texttt{/etc/crontab} : crontab syst\`eme (format avec utilisateur) % Crontab système
    \item \texttt{/etc/cron.d/} : scripts cron additionnels % Scripts additionnels
    \item \texttt{/etc/cron.hourly/} : scripts horaires % Scripts horaires
    \item \texttt{/etc/cron.daily/} : scripts quotidiens % Scripts quotidiens
    \item \texttt{/etc/cron.weekly/} : scripts hebdomadaires % Scripts hebdomadaires
    \item \texttt{/etc/cron.monthly/} : scripts mensuels % Scripts mensuels
\end{itemize}

\subsection{at : t\^aches ponctuelles}
% at : exécution unique à date/heure spécifiée
% atd : démon nécessaire (service atd)
\begin{itemize}
    \item \texttt{at 14:30} : ex\'ecuter \`a 14h30 aujourd'hui % Heure aujourd'hui
    \item \texttt{at now + 1 hour} : dans 1 heure % Dans 1 heure
    \item \texttt{at 3pm tomorrow} : demain \`a 15h % Demain
    \item \texttt{at 10:00 2024-12-25} : date sp\'ecifique % Date spécifique
    \item \texttt{atq} : lister t\^aches planifi\'ees % Lister tâches
    \item \texttt{atrm num\'ero} : supprimer t\^ache % Supprimer tâche
    \item \texttt{at -l} : \'equivalent de atq % Lister
    \item \texttt{at -d num\'ero} : \'equivalent de atrm % Supprimer
\end{itemize}

\subsection{systemd timers}
% systemd timers : alternative moderne à cron

    \item Unit\'es : \texttt{.timer} (planification) et \texttt{.service} (t\^ache) % Deux unités
    \item \texttt{systemctl list-timers} : lister timers actifs % Lister timers

    \item \texttt{systemctl enable timer.timer} : activer au boot % Au démarrage
    \item Avantages : d\'ependances, logs int\'egr\'es, activation conditionnelle % Avantages
\end{itemize}

\section{Variables d'Environnement et Shell}

\subsection{Variables d'environnement}

\subsubsection{D\'efinition}
% Variables d'environnement : configuration accessible aux processus
% Héritées par processus enfants
Les variables d'environnement sont des valeurs stock\'ees et accessibles par tous les processus.

\subsubsection{Variables importantes}
\begin{itemize}
    \item \texttt{PATH} : chemins de recherche des ex\'ecutables % Chemins exécutables
    \item \texttt{HOME} : r\'epertoire personnel % Home directory
    \item \texttt{USER} : nom d'utilisateur actuel % Utilisateur
    \item \texttt{UID} : identifiant utilisateur % User ID
    \item \texttt{SHELL} : shell par d\'efaut % Shell par défaut
    \item \texttt{PWD} : r\'epertoire courant % Current directory
    \item \texttt{PS1} : prompt principal % Prompt shell
    \item \texttt{PS2} : prompt secondaire (continuation) % Prompt continuation
    \item \texttt{LANG} : langue et encodage % Locale
    \item \texttt{EDITOR} : \'editeur par d\'efaut % Éditeur
\end{itemize}

\subsubsection{Commandes}
\begin{itemize}
    \item \texttt{echo \$VARIABLE} : afficher variable % Afficher
    \item \texttt{export VARIABLE=valeur} : d\'efinir et exporter % Définir et exporter
    \item \texttt{VARIABLE=valeur commande} : variable pour commande uniquement % Variable locale
    \item \texttt{env} : lister toutes les variables % Lister toutes
    \item \texttt{set} : lister variables shell et environnement % Variables shell
    \item \texttt{unset VARIABLE} : supprimer variable % Supprimer
\end{itemize}

\subsubsection{Fichiers de configuration}
\begin{itemize}
    \item \texttt{/etc/profile} : configuration globale (tous shells) % Global, tous shells
    \item \texttt{/etc/bashrc} : configuration bash globale % Bash global
    \item \texttt{\textasciitilde/.bashrc} : configuration bash utilisateur (non-login) % Bash utilisateur
    \item \texttt{\textasciitilde/.bash\_profile} : configuration bash login % Bash login
    \item \texttt{\textasciitilde/.profile} : configuration shell login % Shell login
\end{itemize}

\subsection{Redirections et pipes}

\subsubsection{Redirections}
\begin{itemize}
    \item \texttt{commande > fichier} : rediriger sortie standard (\'ecrase) % Écraser fichier
    \item \texttt{commande >> fichier} : rediriger sortie standard (ajoute) % Ajouter
    \item \texttt{commande < fichier} : rediriger entr\'ee standard % Entrée depuis fichier
    \item \texttt{commande 2> fichier} : rediriger erreurs % Erreurs
    \item \texttt{commande > fichier 2>\&1} : rediriger sortie et erreurs % Sortie et erreurs
    \item \texttt{commande \&> fichier} : rediriger sortie et erreurs (bash) % Toutes sorties
    \item \texttt{commande 2>/dev/null} : ignorer erreurs % Ignorer erreurs
    \item \texttt{commande > fichier 2>\&1} : sortie et erreurs dans fichier % Redirection combinée
\end{itemize}

\subsubsection{Pipes}
\begin{itemize}
    \item \texttt{commande1 | commande2} : sortie commande1 vers entr\'ee commande2 % Pipe standard
    \item \texttt{commande1 | tee fichier | commande2} : sauvegarder et continuer % Tee : sauvegarder
    \item \texttt{commande | grep pattern} : filtrer sortie % Filtrer
    \item \texttt{commande | sort} : trier sortie % Trier
    \item \texttt{commande | head -n 10} : 10 premi\`eres lignes % Premières lignes
    \item \texttt{commande | tail -n 10} : 10 derni\`eres lignes % Dernières lignes
\end{itemize}

\section{Recherche de Fichiers}

% ------------------------------------------------------------
% Sous-section : find
% ------------------------------------------------------------
% find : recherche puissante et flexible. Critères multiples.
% Exécution de commandes sur résultats.

\subsection{find : recherche avanc\'ee}

\subsubsection{Syntaxe de base}
% find : recherche récursive puissante
% Syntaxe : find [chemin] [critères] [actions]
\texttt{find [chemin] [crit\`eres] [actions]}

\subsubsection{Crit\`eres de recherche}
\begin{itemize}
    \item \texttt{find . -name "*.txt"} : par nom (sensible \`a la casse) % Par nom
    \item \texttt{find . -iname "*.txt"} : par nom (insensible \`a la casse) % Insensible casse
    \item \texttt{find / -type f} : fichiers uniquement % Fichiers
    \item \texttt{find / -type d} : r\'epertoires uniquement % Répertoires
    \item \texttt{find . -size +100M} : taille sup\'erieure \`a 100 Mo % Par taille
    \item \texttt{find . -size -1k} : taille inf\'erieure \`a 1 Ko % Taille petite
    \item \texttt{find . -mtime -7} : modifi\'e dans les 7 derniers jours % Par date modification
    \item \texttt{find . -mtime +30} : modifi\'e il y a plus de 30 jours % Ancien
    \item \texttt{find . -user utilisateur} : appartenant \`a utilisateur % Par propriétaire
    \item \texttt{find . -group groupe} : appartenant \`a groupe % Par groupe
    \item \texttt{find . -perm 644} : permissions exactes % Par permissions
    \item \texttt{find . -perm -u+x} : avec setuid % Permissions spéciales
\end{itemize}

\subsubsection{Actions}
\begin{itemize}
    \item \texttt{find . -name "*.log" -delete} : supprimer fichiers trouv\'es % Supprimer
    \item \texttt{find . -name "*.txt" -exec ls -l \{\} \textbackslash;} : ex\'ecuter commande % Exécuter commande
    \item \texttt{find . -name "*.txt" -ok rm \{\} \textbackslash;} : confirmation avant action % Avec confirmation
    \item \texttt{find . -name "*.txt" -print0 | xargs -0 rm} : passer \`a xargs % Avec xargs
\end{itemize}

\subsubsection{Exemples pratiques}
\begin{itemize}
    \item \texttt{find /var/log -name "*.log" -mtime +30 -delete} : nettoyer vieux logs % Nettoyage logs
    \item \texttt{find /home -type f -size +100M} : gros fichiers % Gros fichiers
    \item \texttt{find . -name "core" -type f} : fichiers core % Fichiers core
    \item \texttt{find /etc -name "*.conf" -exec grep -l "pattern" \{\} \textbackslash;} : chercher pattern % Recherche dans fichiers
\end{itemize}

\subsection{locate : recherche rapide}
% locate : recherche via base de données pré-indexée
% Plus rapide que find, mais base doit être à jour
\begin{itemize}

    \item \texttt{locate -i fichier} : insensible \`a la casse % Insensible casse

    \item \texttt{updatedb} : mettre \`a jour base de donn\'ees (root) % Mettre à jour base
    \item Avantage : tr\`es rapide % Performance
    \item Inconv\'enient : base peut \^etre obsol\`ete % Base peut être obsolète
\end{itemize}
